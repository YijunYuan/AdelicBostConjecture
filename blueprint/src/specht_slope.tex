\chapter{Determinant--Specht bundles and the geometric slope inequality}

\begin{lemma}[The integral Specht action and its complex metric]
\label{lem:arithmetic-specht-action}
\uses{def:rectangular-specht}
\source{bost-specht-manuscript:page-0004,bost-specht-manuscript:page-0009}
\lean{BostConjecture.Arithmetic.Specht.integralRepresentation,BostConjecture.Arithmetic.Specht.complexRepresentationEquiv,BostConjecture.Arithmetic.Specht.complexRepresentation_irreducible,BostConjecture.Arithmetic.Specht.latticeComplexCore_action_inner,BostConjecture.Arithmetic.Specht.scalarExtensionRepresentationEquiv}
\leanok
Let $L$ be the integral orbit lattice of a polytabloid of shape $\lambda$.
Position permutations define a representation of $\mathfrak S_d$ on $L$.
Its complexification is equivariantly isomorphic to the complex Specht
representation of shape $\lambda$, and is irreducible. The Hermitian form
induced from orthonormal word coordinates is invariant under this action.
For every commutative ring $R$ with a homomorphism $R\to\mathbb C$, the
canonical identification
$\mathbb C\otimes_R(R\otimes_{\mathbb Z}L)\simeq
\mathbb C\otimes_{\mathbb Z}L$ intertwines the two permutation actions.
\end{lemma}
\begin{proof}
The orbit span is stable under position permutations: applying $g$ to
$h e_t$ gives $(gh)e_t$, and the action preserves integer linear
combinations. Restricting the word representation therefore gives the
asserted integral representation. The complexified inclusion identifies
its image with the complex span of these same orbit vectors, namely the
Specht module. It is injective because complex scalar extension is flat
over $\mathbb Z$. The identification intertwines each permutation on
integral generators and hence on their complex span. Irreducibility is
the characteristic-zero irreducibility theorem for complex Specht modules.
In word coordinates the action sends the coefficient at $w$ to that at
$w\circ g$. This is a permutation of the finite coordinate set, so it
preserves $\sum_w\overline{x_w}y_w$. Finally, cancellation of scalar
extension sends $z\otimes(r\otimes x)$ to $z\,r\otimes x$; applying
a permutation to $x$ commutes with this formula. Additivity extends the
identity to all tensors.
\end{proof}

\begin{lemma}[The arithmetic Specht lattice is semistable]
\label{lem:arithmetic-specht-semistable}
\uses{lem:arithmetic-specht-action,lem:irreducible-semistable}
\source{bost-specht-manuscript:page-0009}
\lean{BostConjecture.Arithmetic.Specht.latticeMetric,BostConjecture.Arithmetic.Specht.latticeMetric_action_inner,BostConjecture.Arithmetic.Specht.latticeMetric_isSemistable}
\leanok
For a number field $K$, equip $\Ocal_K\otimes_{\mathbb Z}L$ with the
word Gram form at every complex embedding, using the canonical
scalar-extension cancellation. This is a Hermitian vector bundle, and
its actual integral permutation representation is isometric in every
archimedean fiber. The bundle is semistable.
\end{lemma}
\begin{proof}
The integral lattice $L$ is finite free and nonzero, so its scalar
extension is finite free of the same positive rank. Cancellation of
scalar extension transports the positive-definite word form to each
complex fiber. For a pure integral tensor $r\otimes x$, its word
coefficients at $\sigma$ are $\sigma(r)x_w$. Replacing $\sigma$ by its
conjugate conjugates these coefficients because $x_w$ is an integer.
The same identity holds for sums, and shows that the resulting forms
satisfy conjugation compatibility. The equivariance and invariance in
Lemma~\ref{lem:arithmetic-specht-action} give the asserted isometric
action in every fiber. At any one complex embedding the representation
is irreducible by that lemma. The irreducible isometric action criterion
therefore gives semistability.
\end{proof}

\begin{lemma}[The determinant-twisted permutation representation]
\label{lem:determinant-specht-representation}
\uses{lem:arithmetic-specht-action,lem:projective-determinant-inclusion,lem:determinant-free-base-change-square}
\source{bost-specht-manuscript:page-0008,bost-specht-manuscript:page-0009}
\lean{BostConjecture.Arithmetic.Specht.determinantRepresentation,BostConjecture.Arithmetic.Specht.projectiveDeterminantMap_equivariant,BostConjecture.Arithmetic.Specht.determinantRepresentationEquiv,BostConjecture.Arithmetic.Specht.determinantRepresentation_irreducible_of_rankAtStalk}
\leanok
Let $A$ be a finite projective $R$-module of constant rank $a$, let
$d=ak$ with $k>0$, and let $R\to\mathbb C$ be a ring homomorphism.
The permutation action on
$(\bigotimes_{q=1}^k\bigwedge^a A)\otimes_R(R\otimes_{\mathbb Z}L_{a,d})$
is trivial on the determinant factor and is the orbit action on the lattice
factor. The projective determinant map intertwines this action with
position permutations. Its complexification is irreducible and is
equivariantly isomorphic to the complex rectangular Specht representation.
\end{lemma}
\begin{proof}
The tensor product of the trivial determinant action and the integral
lattice action is a representation on the displayed module. Equivariance
of the projective determinant map follows from its position
permutation identity. Base change for exterior and
tensor powers identifies the complex fiber with
$(\bigotimes_{q=1}^k\bigwedge^a A_{\mathbb C})\otimes_{\mathbb C}
(\mathbb C\otimes_{\mathbb Z}L_{a,d})$.
Constant stalk rank implies $\dim_{\mathbb C}A_{\mathbb C}=a$:
apply the stalk-rank base-change identity at the zero prime of
$\mathbb C$. Choose a basis of $A_{\mathbb C}$. Its top wedge gives
an isomorphism $\bigwedge^a A_{\mathbb C}\simeq\mathbb C$, and the
$k$-fold tensor of this isomorphism trivializes the determinant factor.
This operation commutes with permutations because their action on that
factor is trivial. Cancellation of scalar extension and the equivariant
Specht identification in Lemma~\ref{lem:arithmetic-specht-action} give the
claimed equivariant isomorphism. Irreducibility follows from that of the
complex Specht representation. The basis was chosen only after extension
to $\mathbb C$; no basis of $A$ over $R$ is required.
\end{proof}

\begin{lemma}[Semistability of the canonical determinant--Specht metric]
\label{lem:determinant-specht-semistable}
\uses{lem:determinant-specht-representation,lem:projective-determinant-inclusion,lem:irreducible-semistable}
\source{bost-specht-manuscript:page-0008,bost-specht-manuscript:page-0009}
\lean{BostConjecture.Arithmetic.Specht.determinantMetric,BostConjecture.Arithmetic.Specht.determinantMetric_action_inner,BostConjecture.Arithmetic.Specht.determinantMetric_isSemistable}
\leanok
For a finite projective Hermitian $\Ocal_K$-module $A$ of rank $a$ and
$d=ak$ with $k>0$, give $\mathcal S_{a,d}(A)$ the metric induced by
its canonical inclusion in $A^{\otimes d}$. Its actual permutation
representation is isometric at every complex embedding, and this
Hermitian bundle is semistable.
\end{lemma}
\begin{proof}
The canonical inclusion is injective over $\Ocal_K$, since this ring
is flat over $\mathbb Z$. Its pullback of the Hilbert tensor metric is
positive definite on each complex fiber and compatible with conjugation.
The inclusion intertwines the integral position actions, and canonical
tensor base change preserves this identity in every complex fiber.
Position permutations preserve the Hilbert tensor inner product: on
pure tensors the product of the factor inner products is merely
reindexed, and sesquilinearity extends the identity to all tensors.
The pullback metric is therefore invariant under the actual integral
permutation action. At one complex embedding that action is irreducible
by Lemma~\ref{lem:determinant-specht-representation}; in particular the
complex fiber, and hence the generic fiber of equal dimension, is
nonzero. The irreducible isometric action criterion gives semistability.
\end{proof}

\begin{lemma}[The canonical determinant--Specht fiber metric]
\label{lem:determinant-specht-fiber-metric}
\uses{lem:determinant-specht-semistable,lem:projective-determinant-isometry}
\source{bost-specht-manuscript:page-0008,bost-specht-manuscript:page-0009}
\lean{BostConjecture.Arithmetic.Specht.determinantMetric_fiber_inner}
\leanok
At every complex embedding, canonical exterior and tensor base change
identify the induced metric on $\mathcal S_{a,d}(A)$ with the tensor
product of the $k$ determinant metrics and the canonical complex lattice
Gram form.
\end{lemma}
\begin{proof}
Choose an orthonormal basis of $A_{\sigma}$. The projective determinant
map base-changes to its free formula in this basis. Its image metric is
the Hilbert tensor form. By Lemma~\ref{lem:projective-determinant-isometry},
that formula pulls this form back to the tensor product of the Gram
determinant forms and the lattice form under the same canonical
base-change equivalence. This is precisely the defining pullback metric
of $\mathcal S_{a,d}(A)$. There is no further normalization of the
determinant tensor: its unnormalized norm is already present in the
polytabloid lattice form.
\end{proof}

\begin{lemma}[Independent determinant--Specht factors]
\label{lem:determinant-specht-family-semistable}
\uses{lem:determinant-specht-semistable,lem:independent-irreducible-semistable}
\source{bost-specht-manuscript:page-0009}
\lean{BostConjecture.Arithmetic.Specht.determinantMetric_piTensor_and_dual_isSemistable}
\leanok
For a finite family of finite projective Hermitian modules $A_j$ of
constant positive ranks $a_j$, and a common $d=a_jk_j$ with $k_j>0$,
both $\bigotimes_j\mathcal S_{a_j,d}(A_j)$ and its Hermitian dual are
semistable.
\end{lemma}
\begin{proof}
Each actual determinant--Specht representation is irreducible at the
same chosen complex embedding and is isometric at every embedding, by
Lemma~\ref{lem:determinant-specht-semistable} and its proof. Each factor
has positive generic rank. Apply the finite independent action theorem
to the product of the permutation groups, using the canonical Hilbert
tensor metric. It gives semistability of both the tensor bundle and its
Hermitian dual, with no arithmetic semistability hypothesis on the $A_j$.
\end{proof}

\begin{lemma}[Determinant-isotypic integral modules]
\label{lem:determinant-isotypic}
\uses{def:rectangular-specht,lem:arithmetic-specht-action,lem:arithmetic-specht-semistable,lem:rectangular-invariants,lem:determinant-twisted-free-inclusion,lem:determinant-free-base-change,lem:determinant-free-naturality,lem:tensor-base-change-equivalence,lem:determinant-free-base-change-square,lem:linear-map-localization-descent,lem:determinant-chart-tower,lem:finite-projective-frames,lem:projective-determinant-inclusion,lem:degree-calculus}
\source{bost-specht-manuscript:page-0008,bost-specht-manuscript:page-0009}
\notready
For a rank-$a$ projective $\Ocal$-module $A$ and $d=ak>0$, there is
a canonical injection
\[
 \iota_A:\mathcal S_{a,d}(A):=
 (\det A)^{\otimes k}\otimes_\mathbb Z L_{a,d}\longrightarrow A^{\otimes d}.
\]
It is isometric for the determinant and canonical Specht metrics and
satisfies
\[
 \slope(\mathcal S_{a,d}(\overline A))=d\slope(\overline A)-c_a(d).
\]
Its generic fiber is the full $\SL(A_K)$-invariant subspace of
$A_K^{\otimes d}$, with absolutely irreducible isometric $\mathfrak S_d$
action.
\end{lemma}
\begin{proof}
In a free module, a rectangular polytabloid is a permutation of a
tensor product of $k$ unnormalized determinant tensors. Thus every
change-of-frame matrix $g$ acts on it by $(\det g)^k$. For a local
frame $b_1,\ldots,b_a$ the formula
\[
 (b_1\wedge\cdots\wedge b_a)^{\otimes k}\otimes x
 \longmapsto b^{\otimes d}x
\]
is consequently independent of the frame. Choose a finite principal
cover on which $A$ is free. Exterior-power and tensor base change
identify the local sources and targets with localizations of the
intrinsic global modules. Scalar-extension compatibility and frame
independence make the local maps agree on every overlap. Descent of
linear maps therefore gives a unique global map over $\Ocal$.
The local inclusion of the Specht lattice into the word lattice stays
injective after scalar extension because $\Ocal$ and its localizations
are torsion-free, hence flat, over $\mathbb Z$. This proves injectivity.

At each archimedean place choose an orthonormal frame. Tensor words are
orthonormal, as are the corresponding row tabloids, and the determinant
vector of the frame has norm one. The formula is therefore an isometry.
In particular, no division by a factorial is made: the norm
$\sqrt{a!}$ of an unnormalized determinant tensor already belongs to
the polytabloid's metric.

If $f=\rk_\mathbb Z L_{a,d}$, a basis wedge of
$\Ocal\otimes_\mathbb Z L_{a,d}$ has norm $\covol L_{a,d}$ at every
embedding and generates its determinant module. Its normalized degree
is $-\log\covol L_{a,d}$. The line-twist identity gives the slope
$k\degn(\det A)-f^{-1}\log\covol L_{a,d}$, which is the asserted
formula. The rectangular invariant-space theorem identifies the
generic fiber and its absolute irreducibility. Permutation of tensor
positions preserves the lattice and the ambient Hilbert tensor norm,
and commutes with changes of frame, proving the isometric action.
\end{proof}

\begin{theorem}[Geometric slope inequality]
\label{thm:geometric-slope}
\uses{lem:contraction-nonzero,lem:canonical-contraction,lem:determinant-isotypic,lem:irreducible-semistable,lem:degree-calculus,thm:specht-sublinear-covolume}
\source{bost-specht-manuscript:page-0009,bost-specht-manuscript:page-0010}
\notready
For arbitrary positive-rank Hermitian bundles $E_1,\ldots,E_N$ and
a nonzero submodule $V\subseteq\bigotimes_j E_j$, geometric
semistability of the Pl\"ucker line of $V_K$ implies
\[
 \slope(\overline V)\leq\sum_j\slope(\overline E_j).
\]
The factors need not be arithmetically semistable.
\end{theorem}
\begin{proof}
Take the unbounded sequence $d=\ell d_0$ from the nonvanishing lemma.
The kernel of $\Psi_{d,K}$ is invariant for the simultaneous
$\mathfrak S_d$ action. Absolute irreducibility of $B_{d,K}$ and
nonvanishing force that kernel to be zero. Torsion-freeness gives
injectivity over $\Ocal$. Let $J_d$ be the actual image, with metric
induced from $D_d^\vee$. The isomorphism of modules $B_d\to J_d$
is contracting, so $\slope(B_d)\leq\slope(J_d)$.

Each factor of $D_d$ has absolutely irreducible isometric permutation
action. Their tensor product is absolutely irreducible for the
independent product of permutation groups. Its dual is semistable by
the irreducible-action lemma. Thus, including when $J_d$ is not
saturated, $\slope(J_d)\leq\slope(D_d^\vee)$. The determinant and
average tensor-slope formulas give
\[
 d\slope(V)-c_s(d)\leq\slope(J_d)
 \leq d\sum_j\slope(E_j)+\sum_jc_{a_j}(d).
\]
Divide by $d$ and let $\ell\to\infty$. The sublinear Specht covolume
theorem makes every error term tend to zero, giving the claim.
\end{proof}
