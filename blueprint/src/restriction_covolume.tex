\section{Restriction of scalars and arithmetic covolume}

Let $K$ be a number field, $\Ocal=\mathcal O_K$ its ring of integers,
$D=[K:\mathbb Q]$, and $\Delta_K=|\operatorname{disc}(K)|$.
At an infinite place $v$, put $w_v=1$ for a real place and $w_v=2$
for a complex place. Let $A$ be a finite projective $\Ocal$-module
with positive definite Hermitian metrics at the complex embeddings,
compatible with conjugation, as in Definition~\ref{def:hermitian-metric}.
The normalized degree is Definition~\ref{def:intrinsic-determinant-degree}.
Covolume below uses the Euclidean volume of the displayed real trace metric.

\begin{lemma}[The real trace space and integral image]
\label{lem:real-trace-lattice}
\uses{def:hermitian-metric,lem:bounded-integral-sections,lem:integral-restriction-rank}
\source{bost-specht-manuscript:page-0011}
\lean{BostConjecture.Arithmetic.HermitianMetric.traceCore,BostConjecture.Arithmetic.HermitianMetric.traceSection_injective,BostConjecture.Arithmetic.HermitianMetric.traceLattice_discreteTopology,BostConjecture.Arithmetic.HermitianMetric.traceLattice_isZLattice,BostConjecture.Arithmetic.HermitianMetric.realTensorEquiv,BostConjecture.Arithmetic.HermitianMetric.realTensorEquiv_tmul,BostConjecture.Arithmetic.HermitianMetric.traceSpace_finrank}\leanok
Let $j:A\longrightarrow\prod_{\sigma:K\hookrightarrow\mathbb C} A_\sigma$
be the simultaneous section map and let $E=\operatorname{span}_{\mathbb R}j(A)$.
The pairing
\[
 (x,y)_{\mathrm{tr}}=\sum_\sigma\operatorname{Re}\langle x_\sigma,y_\sigma\rangle_\sigma
\]
is a positive definite real inner product on $E$. The map $j$ is injective,
$j(A)$ is a full discrete integer lattice, and the scalar extension
$\mathbb R\otimes_{\mathbb Z}A\longrightarrow E$, $r\otimes a\mapsto rj(a)$,
is an isomorphism. In particular $\dim_{\mathbb R}E=D\rk_\Ocal A$.
\end{lemma}
\begin{proof}
Real bilinearity and symmetry follow from the Hermitian pairing in each
fiber. Each summand of $(x,x)_{\mathrm{tr}}$ is nonnegative; their sum
vanishes only if every coordinate is zero. Scalar extension to any complex
embedding is injective on the projective module $A$, so $j$ is injective.
The square of the norm of $j(a)$ is $\sum_\sigma\|a\|_\sigma^2$.
Consequently a bounded ball meets $j(A)$ in finitely many points by
Lemma~\ref{lem:bounded-integral-sections}. Removing the finitely many
nonzero points from the unit ball gives a neighborhood of zero meeting
$j(A)$ only at zero; translation proves discreteness. Fullness is the
definition of $E$.

For completeness, discreteness also proves the required independence over
$\mathbb R$. Choose an integer basis of $j(A)$ and a maximal real
independent subset $e_1,\dots,e_s$ of that basis. Every remaining basis
element $v$ has all integer multiples reducible modulo
$\sum_i\mathbb Ze_i$ into its bounded fundamental parallelepiped.
These remainders lie in the discrete subgroup $j(A)$, so only finitely many
occur. Two distinct multiples have the same remainder, giving
$nv\in\sum_i\mathbb Ze_i$ for a nonzero integer $n$, contrary to integer
independence. Thus the whole integer basis is a real basis. Its scalar
extension is the asserted isomorphism, which is canonical because its
formula on pure tensors determines it. Its dimension is the integer rank
of $A$, equal to $D\rk_\Ocal A$ by
Lemma~\ref{lem:integral-restriction-rank}.
\end{proof}

\begin{lemma}[Conjugation and the trace metric]
\label{lem:trace-metric-weights}
\uses{lem:real-trace-lattice,def:hermitian-metric}
\source{bost-specht-manuscript:page-0011}
\lean{BostConjecture.Arithmetic.HermitianMetric.traceInclusion_conjugate,BostConjecture.Arithmetic.HermitianMetric.trace_inner_eq_sum_places,BostConjecture.Arithmetic.HermitianMetric.trace_norm_sq_eq_sum_places}\leanok
Every $x\in E$ satisfies $x_{\bar\sigma}=\overline{x_\sigma}$ under
canonical fiber conjugation. For $x,y\in E$,
\[
 (x,y)_{\mathrm{tr}}=\sum_{v\mid\infty} w_v
   \operatorname{Re}\langle x_v,y_v\rangle_v,
 \qquad \|x\|_{\mathrm{tr}}^2=\sum_{v\mid\infty}w_v\|x_v\|_v^2.
\]
\end{lemma}
\begin{proof}
Conjugation takes $z\otimes a$ to $\bar z\otimes a$, so the component
identity holds for each $j(a)$. Both sides are real linear in $x$, hence
it holds on their real span $E$. Compatibility of the metrics with fiber
conjugation implies that the pairings at $\sigma$ and $\bar\sigma$ are
complex conjugates. Their real parts coincide. Group the finite sum over
embeddings by infinite place: a real place has one embedding and a complex
place has exactly two. This gives the first equality; setting $y=x$
gives the squared norm equality.
\end{proof}

\begin{lemma}[Euclidean Haar covolume and the trace Gram matrix]
\label{lem:trace-covolume-gram}
\uses{lem:real-trace-lattice,lem:trace-metric-weights,lem:degree-rank-zero}
\source{bost-specht-manuscript:page-0011}
\lean{BostConjecture.Arithmetic.HermitianMetric.traceCovolume_pos,BostConjecture.Arithmetic.HermitianMetric.traceCovolume_sq_eq_det,BostConjecture.Arithmetic.HermitianMetric.traceCovolume_eq_discr_degree_of_rank_zero}\leanok
For every integer basis $(b_i)$ of $A$, the Euclidean Haar covolume $c$ of
$j(A)$ is positive and satisfies
\[
 c^2=\det\left(\sum_\sigma
   \operatorname{Re}\langle b_i,b_j\rangle_\sigma\right)_{i,j}.
\]
If $a=0$, then $c=1=\Delta_K^{a/2}\exp(-D\degn\overline A)$.
\end{lemma}
\begin{proof}
By Lemma~\ref{lem:real-trace-lattice}, $(j(b_i))$ is a real basis of $E$.
Choose an orthonormal basis and let $T$ be the coordinate matrix of
$(j(b_i))$ in it. The half-open unit parallelepiped has Euclidean Haar
volume one. Change of variables shows that the fundamental
parallelepiped of $j(A)$ has volume $c=|\det T|>0$. Its Gram matrix is
$T^{\mathsf T}T$, whose determinant is $(\det T)^2=c^2$.
The trace-pairing definition gives the displayed entries. In rank zero,
the integer basis is empty and its Gram determinant is one, so positivity
gives $c=1$. The integral empty wedge generates the determinant line and
has norm one at every embedding. Its quotient has cardinality one, so
the intrinsic determinant degree is zero. The claimed rank-zero formula
follows.
\end{proof}

\begin{lemma}[Finite-index change of trace covolume]
\label{lem:trace-covolume-index}
\uses{lem:trace-covolume-gram,lem:real-trace-lattice}
\source{bost-specht-manuscript:page-0011}
\lean{BostConjecture.Arithmetic.HermitianMetric.traceCovolume_submodule}\leanok
If $P\subseteq A$ has the same generic rank as $A$, equip $P$ with the
induced Hermitian metrics. Then
\[
 \covol_{\mathbb Z}(P)=\#(A/P)\covol_{\mathbb Z}(A).
\]
\end{lemma}
\begin{proof}
The image $j(P)$ is a subgroup of the discrete group $j(A)$ and hence
discrete. The real span of a discrete integer subgroup has dimension its
integer rank, by the basis argument in Lemma~\ref{lem:real-trace-lattice}.
The equal generic ranks therefore imply that $j(P)$ spans the whole real
trace space of $A$. For any integer basis of $P$, its Gram matrix using
the induced metrics agrees with its Gram matrix in that ambient trace
space. Lemma~\ref{lem:trace-covolume-gram} and positivity thus identify
its intrinsic covolume with the ambient lattice covolume.
The change-of-basis matrix from an integer basis of $j(P)$ to an integer
basis of $j(A)$ is an integral square matrix with nonzero determinant.
Its absolute determinant equals the index, by integer Smith normal form,
and multiplies the volume of the fundamental parallelepiped.
Finally the injective section map identifies $A/P$ with $j(A)/j(P)$.
This proves the formula and finiteness of the quotient.
\end{proof}

\begin{lemma}[Restriction of scalars and covolume]
\label{lem:restriction-covolume}\uses{def:hermitian-degree,lem:bounded-integral-sections,lem:integral-restriction-rank,lem:real-trace-lattice,lem:trace-metric-weights,lem:trace-covolume-gram,lem:trace-covolume-index}
\source{bost-specht-manuscript:page-0011}\notready
Give $A\otimes_{\mathbb Z}\mathbb R$ the Euclidean norm
$\|x\|^2=\sum_v w_v\|x_v\|_v^2$. For $a=\rk_\Ocal A$,
\[
 \covol_\mathbb Z(A)=\Delta_K^{a/2}\exp(-D\degn(\overline A)).
\]
\end{lemma}
\begin{proof}
The underlying abelian group of $A$ is finite free and spans its real
scalar extension. A bounded set for the displayed Euclidean norm bounds
every archimedean section norm. Lemma~\ref{lem:bounded-integral-sections}
therefore makes its intersection with $A$ finite, so $A$ is a full
discrete lattice in this real space.
For $A=\Ocal^a$ with the standard metrics, the Gram determinant of an
integral basis of $\Ocal$ for this real trace metric is $\Delta_K$.
Indeed, the matrix of embeddings gives the trace discriminant; replacing
a conjugate pair by real and imaginary parts and weighting both by $2$
preserves its absolute squared determinant. Orthogonal direct sum gives
the standard covolume $\Delta_K^{a/2}$. If the fiber metrics have Gram
matrices $G_v$, covolume is multiplied by
$\prod_v(\det G_v)^{w_v/2}$, while the definition of degree multiplies
$\exp(-D\degn A)$ by precisely that quantity. This proves the formula
for a free module with arbitrary metrics. For general $A$, choose a
generic basis lying in $A$ and its finite-index free span $A_0$.
Then $\covol(A_0)=[A:A_0]\covol(A)$ and
$D\degn(A)=D\degn(A_0)+\log[A:A_0]$; cancellation proves the claim.
\end{proof}
