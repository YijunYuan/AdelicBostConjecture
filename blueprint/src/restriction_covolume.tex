\section{Restriction of scalars and arithmetic covolume}

Let $K$ be a number field, $\Ocal=\mathcal O_K$ its ring of integers,
$D=[K:\mathbb Q]$, and $\Delta_K=|\operatorname{disc}(K)|$.
At an infinite place $v$, put $w_v=1$ for a real place and $w_v=2$
for a complex place. Let $A$ be a finite projective $\Ocal$-module
with positive definite Hermitian metrics at the complex embeddings,
compatible with conjugation, as in Definition~\ref{def:hermitian-metric}.
The normalized degree is Definition~\ref{def:intrinsic-determinant-degree}.
Covolume below uses the Euclidean volume of the displayed real trace metric.

\begin{lemma}[The real trace space and integral image]
\label{lem:real-trace-lattice}
\uses{def:hermitian-metric,lem:bounded-integral-sections,lem:integral-restriction-rank}
\source{bost-specht-manuscript:page-0011}
\lean{BostConjecture.Arithmetic.HermitianMetric.traceCore,BostConjecture.Arithmetic.HermitianMetric.traceSection_injective,BostConjecture.Arithmetic.HermitianMetric.traceLattice_discreteTopology,BostConjecture.Arithmetic.HermitianMetric.traceLattice_isZLattice,BostConjecture.Arithmetic.HermitianMetric.realTensorEquiv,BostConjecture.Arithmetic.HermitianMetric.realTensorEquiv_tmul,BostConjecture.Arithmetic.HermitianMetric.traceSpace_finrank}\leanok
Let $j:A\longrightarrow\prod_{\sigma:K\hookrightarrow\mathbb C} A_\sigma$
be the simultaneous section map and let $E=\operatorname{span}_{\mathbb R}j(A)$.
The pairing
\[
 (x,y)_{\mathrm{tr}}=\sum_\sigma\operatorname{Re}\langle x_\sigma,y_\sigma\rangle_\sigma
\]
is a positive definite real inner product on $E$. The map $j$ is injective,
$j(A)$ is a full discrete integer lattice, and the scalar extension
$\mathbb R\otimes_{\mathbb Z}A\longrightarrow E$, $r\otimes a\mapsto rj(a)$,
is an isomorphism. In particular $\dim_{\mathbb R}E=D\rk_\Ocal A$.
\end{lemma}
\begin{proof}
Real bilinearity and symmetry follow from the Hermitian pairing in each
fiber. Each summand of $(x,x)_{\mathrm{tr}}$ is nonnegative; their sum
vanishes only if every coordinate is zero. Scalar extension to any complex
embedding is injective on the projective module $A$, so $j$ is injective.
The square of the norm of $j(a)$ is $\sum_\sigma\|a\|_\sigma^2$.
Consequently a bounded ball meets $j(A)$ in finitely many points by
Lemma~\ref{lem:bounded-integral-sections}. Removing the finitely many
nonzero points from the unit ball gives a neighborhood of zero meeting
$j(A)$ only at zero; translation proves discreteness. Fullness is the
definition of $E$.

For completeness, discreteness also proves the required independence over
$\mathbb R$. Choose an integer basis of $j(A)$ and a maximal real
independent subset $e_1,\dots,e_s$ of that basis. Every remaining basis
element $v$ has all integer multiples reducible modulo
$\sum_i\mathbb Ze_i$ into its bounded fundamental parallelepiped.
These remainders lie in the discrete subgroup $j(A)$, so only finitely many
occur. Two distinct multiples have the same remainder, giving
$nv\in\sum_i\mathbb Ze_i$ for a nonzero integer $n$, contrary to integer
independence. Thus the whole integer basis is a real basis. Its scalar
extension is the asserted isomorphism, which is canonical because its
formula on pure tensors determines it. Its dimension is the integer rank
of $A$, equal to $D\rk_\Ocal A$ by
Lemma~\ref{lem:integral-restriction-rank}.
\end{proof}

\begin{lemma}[Conjugation and the trace metric]
\label{lem:trace-metric-weights}
\uses{lem:real-trace-lattice,def:hermitian-metric}
\source{bost-specht-manuscript:page-0011}
\lean{BostConjecture.Arithmetic.HermitianMetric.traceInclusion_conjugate,BostConjecture.Arithmetic.HermitianMetric.trace_inner_eq_sum_places,BostConjecture.Arithmetic.HermitianMetric.trace_norm_sq_eq_sum_places}\leanok
Every $x\in E$ satisfies $x_{\bar\sigma}=\overline{x_\sigma}$ under
canonical fiber conjugation. For $x,y\in E$,
\[
 (x,y)_{\mathrm{tr}}=\sum_{v\mid\infty} w_v
   \operatorname{Re}\langle x_v,y_v\rangle_v,
 \qquad \|x\|_{\mathrm{tr}}^2=\sum_{v\mid\infty}w_v\|x_v\|_v^2.
\]
\end{lemma}
\begin{proof}
Conjugation takes $z\otimes a$ to $\bar z\otimes a$, so the component
identity holds for each $j(a)$. Both sides are real linear in $x$, hence
it holds on their real span $E$. Compatibility of the metrics with fiber
conjugation implies that the pairings at $\sigma$ and $\bar\sigma$ are
complex conjugates. Their real parts coincide. Group the finite sum over
embeddings by infinite place: a real place has one embedding and a complex
place has exactly two. This gives the first equality; setting $y=x$
gives the squared norm equality.
\end{proof}

\begin{lemma}[Euclidean Haar covolume and the trace Gram matrix]
\label{lem:trace-covolume-gram}
\uses{lem:real-trace-lattice,lem:trace-metric-weights,lem:degree-rank-zero}
\source{bost-specht-manuscript:page-0011}
\lean{BostConjecture.Arithmetic.HermitianMetric.traceCovolume_pos,BostConjecture.Arithmetic.HermitianMetric.traceCovolume_sq_eq_det,BostConjecture.Arithmetic.HermitianMetric.traceCovolume_eq_discr_degree_of_rank_zero}\leanok
For every integer basis $(b_i)$ of $A$, the Euclidean Haar covolume $c$ of
$j(A)$ is positive and satisfies
\[
 c^2=\det\left(\sum_\sigma
   \operatorname{Re}\langle b_i,b_j\rangle_\sigma\right)_{i,j}.
\]
If $a=0$, then $c=1=\Delta_K^{a/2}\exp(-D\degn\overline A)$.
\end{lemma}
\begin{proof}
By Lemma~\ref{lem:real-trace-lattice}, $(j(b_i))$ is a real basis of $E$.
Choose an orthonormal basis and let $T$ be the coordinate matrix of
$(j(b_i))$ in it. The half-open unit parallelepiped has Euclidean Haar
volume one. Change of variables shows that the fundamental
parallelepiped of $j(A)$ has volume $c=|\det T|>0$. Its Gram matrix is
$T^{\mathsf T}T$, whose determinant is $(\det T)^2=c^2$.
The trace-pairing definition gives the displayed entries. In rank zero,
the integer basis is empty and its Gram determinant is one, so positivity
gives $c=1$. The integral empty wedge generates the determinant line and
has norm one at every embedding. Its quotient has cardinality one, so
the intrinsic determinant degree is zero. The claimed rank-zero formula
follows.
\end{proof}

\begin{lemma}[Finite-index change of trace covolume]
\label{lem:trace-covolume-index}
\uses{lem:trace-covolume-gram,lem:real-trace-lattice,lem:induced-hermitian-metric,lem:integral-submodule-projective}
\source{bost-specht-manuscript:page-0011}
\lean{BostConjecture.Arithmetic.HermitianMetric.traceCovolume_submodule}\leanok
If $P\subseteq A$ has the same generic rank as $A$, equip $P$ with the
induced Hermitian metrics. Then
\[
 \covol_{\mathbb Z}(P)=\#(A/P)\covol_{\mathbb Z}(A).
\]
\end{lemma}
\begin{proof}
The image $j(P)$ is a subgroup of the discrete group $j(A)$ and hence
discrete. The real span of a discrete integer subgroup has dimension its
integer rank, by the basis argument in Lemma~\ref{lem:real-trace-lattice}.
The equal generic ranks therefore imply that $j(P)$ spans the whole real
trace space of $A$. For any integer basis of $P$, its Gram matrix using
the induced metrics agrees with its Gram matrix in that ambient trace
space. Lemma~\ref{lem:trace-covolume-gram} and positivity thus identify
its intrinsic covolume with the ambient lattice covolume.
The change-of-basis matrix from an integer basis of $j(P)$ to an integer
basis of $j(A)$ is an integral square matrix with nonzero determinant.
Its absolute determinant equals the index, by integer Smith normal form,
and multiplies the volume of the fundamental parallelepiped.
Finally the injective section map identifies $A/P$ with $j(A)/j(P)$.
This proves the formula and finiteness of the quotient.
\end{proof}

\begin{lemma}[Trace covolume in an integral frame]
\label{lem:trace-covolume-free}
\uses{lem:trace-covolume-gram,lem:trace-metric-weights,def:intrinsic-determinant-degree}
\source{bost-specht-manuscript:page-0011}
\lean{BostConjecture.Arithmetic.HermitianMetric.traceCovolume_eq_discr_mul_prod,BostConjecture.Arithmetic.HermitianMetric.traceCovolume_eq_discr_degree_of_basis}\leanok
If $b_1,\dots,b_a$ is an $\Ocal$-basis of $A$ and
$\xi=b_1\wedge\cdots\wedge b_a$, then
\[
 \covol_{\mathbb Z}(A)
 =\Delta_K^{a/2}\prod_{\sigma:K\hookrightarrow\mathbb C}\|\xi\|_\sigma
 =\Delta_K^{a/2}\exp(-D\degn\overline A).
\]
\end{lemma}
\begin{proof}
Choose an integer basis $u_1,\dots,u_D$ of $\Ocal$ and put
$S_{\sigma,t}=\sigma(u_t)$ and
$G_\sigma=(\langle b_i,b_j\rangle_\sigma)_{i,j}$.
The integer basis of $A$ consists of the $u_tb_i$.
The sum of its Hermitian pairings over all embeddings is real, since
conjugate embeddings give conjugate pairings. Its real trace Gram matrix,
viewed over $\mathbb C$, is
\[
 (I_a\otimes S)^*G(I_a\otimes S),
 \qquad G_{(i,\sigma),(j,\tau)}=\delta_{\sigma\tau}(G_\sigma)_{ij}.
\]
Thus $G$ uses the interleaved $(i,\sigma)$ order; a simultaneous row and
column permutation turns it into the usual contiguous block diagonal.
Indeed, its $((i,t),(j,s))$ entry is
$\sum_\sigma\overline{\sigma(u_t)}\sigma(u_s)(G_\sigma)_{ij}$,
which is the pairing of the corresponding integer basis vectors.
The embeddings formula for the field discriminant gives
$|\det S|^2=\Delta_K$. Taking determinants of the displayed factorization
therefore gives
$\covol_{\mathbb Z}(A)^2=\Delta_K^a\prod_\sigma\det G_\sigma$.
The determinant metric gives $\det G_\sigma=\|\xi\|_\sigma^2$,
and all the quantities whose square roots are taken are positive.
This proves the first equality. The section $\xi$ generates the integral
determinant line, so its quotient has cardinality one; the intrinsic
degree formula gives $D\degn\overline A=-\sum_\sigma\log\|\xi\|_\sigma$.
Exponentiating proves the second equality.
\end{proof}

\begin{lemma}[Trace covolume of a projective line]
\label{lem:trace-covolume-line}
\uses{lem:trace-covolume-free,lem:trace-covolume-index,lem:degree-rank-one,lem:line-degree-quotient-formula}
\source{bost-specht-manuscript:page-0011}
\lean{BostConjecture.Arithmetic.HermitianMetric.traceCovolume_eq_discr_degree_of_rank_one}\leanok
For a projective integral module $A$ of generic rank one,
\[
 \covol_{\mathbb Z}(A)=\Delta_K^{1/2}\exp(-D\degn\overline A).
\]
\end{lemma}
\begin{proof}
Choose a nonzero $x\in A$ and set $P=\Ocal x$. Torsion-freeness makes
$a\mapsto ax$ an isomorphism $\Ocal\simeq P$, so $P$ has generic rank
one and $q=\#(A/P)$ is finite and positive. The free-frame formula and
the induced metric give
\[
 q\covol_{\mathbb Z}(A)=\covol_{\mathbb Z}(P)
 =\Delta_K^{1/2}\prod_\sigma\|x\|_\sigma.
\]
In rank one the determinant degree agrees with the intrinsic line degree,
whose section formula is
$D\degn\overline A=\log q-\sum_\sigma\log\|x\|_\sigma$.
Every section norm is positive by injectivity of the fiber section map.
Exponentiation and division by $q$ prove the claim.
\end{proof}

\begin{lemma}[The real form of the product of fibers]
\label{lem:trace-real-form}
\uses{lem:real-trace-lattice,lem:trace-metric-weights,lem:integral-determinant-line}
\source{bost-specht-manuscript:page-0011}
\lean{BostConjecture.Arithmetic.HermitianMetric.traceComplexification_bijective,BostConjecture.Arithmetic.HermitianMetric.mem_archimedeanSpan_iff_conjugate}\leanok
The complexification of $T(A)$ is $W(A)=\prod_\sigma A_\sigma$ by the
extension of the trace inclusion. A family belongs to $T(A)$ exactly when
fiber conjugation carries its $\sigma$ component to its
$\overline\sigma$ component.
\end{lemma}
\begin{proof}
For conjugation-compatible families, the sum of fiber pairings is real:
conjugating this sum permutes its terms. In particular, for $x,y\in T(A)$,
the sum of fiber pairings equals their real trace pairing.
Every element of $\mathbb C\otimes_{\mathbb R}T(A)$ can be written
$1\otimes x+i\otimes y$, by separating real and imaginary parts of each
scalar in a finite tensor sum. The imaginary part of the sum of the
pairings of $y$ with $x+iy$ is $\|y\|^2$. If $x+iy=0$, this forces $y=0$;
injectivity of the trace inclusion then gives $x=0$. Thus complexification
is injective. Both its source and target have complex dimension $Da$, by
the real trace dimension and the equal dimensions of the complex fibers,
so it is bijective. If $z=x+iy$ is conjugation-compatible, the pairing
sum of $y$ with $z$ is real, again forcing $y=0$. Thus $z\in T(A)$.
Conversely every real trace vector is conjugation-compatible by
Lemma~\ref{lem:trace-metric-weights}. This includes rank zero.
\end{proof}

\begin{lemma}[Trace metric and saturated quotients]
\label{lem:trace-quotient-metric}
\uses{lem:trace-real-form,lem:induced-hermitian-metric,lem:quotient-hermitian-metric,lem:exact-determinant-algebra}
\source{bost-specht-manuscript:page-0011}
\lean{BostConjecture.Arithmetic.HermitianMetric.traceMap_pullback_inner,BostConjecture.Arithmetic.HermitianMetric.traceMap_traceLift,BostConjecture.Arithmetic.HermitianMetric.traceLift_inner,BostConjecture.Arithmetic.HermitianMetric.traceLift_orthogonal,BostConjecture.Arithmetic.HermitianMetric.ker_traceMap_mkQ,BostConjecture.Arithmetic.HermitianMetric.traceSplitEquiv}\leanok
Let $P\subset A$ be saturated and $Q=A/P$, with the induced and quotient
Hermitian metrics. The induced map of real trace spaces identifies
$T(P)$ isometrically with its image in $T(A)$. The quotient map identifies
$T(Q)$ isometrically with the Euclidean quotient $T(A)/T(P)$.
\end{lemma}
\begin{proof}
Use Lemma~\ref{lem:trace-real-form} to identify each real trace space with
the conjugation-compatible families in its product of complex fibers.
The fiber maps of inclusion and quotient commute with conjugation and
therefore restrict to the real trace spaces. Inclusion preserves the
trace pairing because every induced fiber pairing is the restriction of
the ambient pairing. The componentwise orthogonal right inverse of the
fiber quotient maps commutes with conjugation: a conjugated orthogonal
representative is the orthogonal representative of the conjugated vector,
by uniqueness. It therefore maps $T(Q)$ into $T(A)$ and is a right inverse.
Consequently the real quotient map is surjective. A real trace vector in
its kernel has each component in the image of the corresponding fiber
of $P$. Choose these preimages; their uniqueness, from injectivity of the
fiber inclusion, makes the preimage family conjugation-compatible.
It therefore belongs to $T(P)$. Conversely the image of $T(P)$ is killed
by the quotient map. Thus the kernel is exactly that image.
The image of this right inverse is orthogonal to $T(P)$, since every
fiber representative is orthogonal to the fiber of $P$. Finally its
trace pairing is precisely the sum of the quotient fiber pairings.
It is thus the orthogonal right inverse defining the Euclidean quotient
metric, which proves both assertions.
\end{proof}

\begin{lemma}[Trace covolume in a saturated exact sequence]
\label{lem:trace-covolume-exact}
\uses{lem:trace-quotient-metric,lem:trace-covolume-gram,lem:integral-restriction-rank}
\source{bost-specht-manuscript:page-0011}
\lean{BostConjecture.Arithmetic.HermitianMetric.traceCovolume_exact}\leanok
With the induced and quotient metrics for a saturated submodule
$P\subset A$, one has
\[
 \covol_{\mathbb Z}(A)=\covol_{\mathbb Z}(P)\covol_{\mathbb Z}(A/P).
\]
\end{lemma}
\begin{proof}
Write $Q=A/P$. Its projectivity gives an integral linear section
$t:Q\to A$ of the quotient map. Choose integer bases $(p_i)$ and $(q_j)$
of $P$ and $Q$. Then $(p_i,t(q_j))$ is an integer basis of $A$:
projecting to $Q$ proves independence and reduces spanning to that of $P$.
Let $s:T(Q)\to T(A)$ be the orthogonal right inverse from
Lemma~\ref{lem:trace-quotient-metric}. For each $j$, the difference
$t(q_j)-s(q_j)$ lies in $T(P)$. Replacing the vectors $t(q_j)$ by $s(q_j)$
is therefore a block triangular real change of basis with identity
diagonal blocks and determinant one. The resulting Gram matrix is block
diagonal: its two blocks are the trace Gram matrices of $(p_i)$ and
$(q_j)$, since $s$ is an isometry and its image is orthogonal to $T(P)$.
Taking determinants gives equality of squared covolumes by
Lemma~\ref{lem:trace-covolume-gram}; their positivity gives the formula.
Empty bases handle zero-dimensional terms by the same calculation.
\end{proof}

\begin{lemma}[Restriction of scalars and covolume]
\label{lem:restriction-covolume}\uses{lem:trace-covolume-gram,lem:trace-covolume-line,lem:trace-covolume-exact,lem:rank-one-saturated-reduction,lem:exact-determinant-degree}
\source{bost-specht-manuscript:page-0011}
\lean{BostConjecture.Arithmetic.HermitianMetric.traceCovolume_eq_discr_degree}\leanok
Give $A\otimes_{\mathbb Z}\mathbb R$ the Euclidean norm
$\|x\|^2=\sum_v w_v\|x_v\|_v^2$. For $a=\rk_\Ocal A$,
\[
 \covol_\mathbb Z(A)=\Delta_K^{a/2}\exp(-D\degn(\overline A)).
\]
\end{lemma}
\begin{proof}
Induct on the generic rank $a$. For $a=0$, the actual lattice has
covolume one and intrinsic degree zero by
Lemma~\ref{lem:trace-covolume-gram}. For $a>0$, choose a saturated
rank-one submodule $L$ by Lemma~\ref{lem:rank-one-saturated-reduction};
the quotient $Q=A/L$ is finite projective of rank $a-1$.
Equip $L$ and $Q$ with the induced and quotient metrics.
Lemmas~\ref{lem:trace-covolume-line} and \ref{lem:trace-covolume-exact},
followed by the induction hypothesis, give
\[
 \covol(A)
 =\Delta_K^{1/2}e^{-D\degn L}
   \Delta_K^{(a-1)/2}e^{-D\degn Q}.
\]
The discriminant is nonzero, hence $\Delta_K>0$, so its real powers
combine to $\Delta_K^{a/2}$. Exact determinant degree additivity gives
$\degn A=\degn L+\degn Q$. Combining the exponentials proves the claim.
\end{proof}
